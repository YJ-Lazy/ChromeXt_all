cmake_minimum_required(VERSION 3.28)
project(chromext_zygisk LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TP "${CMAKE_CURRENT_LIST_DIR}/../deps")
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(Plugin.ImportTableReplace OFF CACHE BOOL "" FORCE)
set(Plugin.Android.BionicLinkerUtil OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${TP}/dobby" "${CMAKE_BINARY_DIR}/dobby")
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
add_subdirectory("${TP}/lsplant/lsplant/src/main/jni" "${CMAKE_BINARY_DIR}/lsplant")
set(XD "${TP}/xdl-wrapper")
set(XC "${XD}/external/xDL/xdl/src/main/cpp")
add_library(chromext SHARED entry.cpp "${XD}/xdl.c"
  "${XC}/xdl_iterate.c" "${XC}/xdl_linker.c" "${XC}/xdl_lzma.c" "${XC}/xdl_util.c")
target_include_directories(chromext PRIVATE "${TP}/zygisk" "${XD}" "${XC}/include")
target_link_libraries(chromext PRIVATE lsplant_static dobby log dl)
target_link_options(chromext PRIVATE "-Wl,-z,max-page-size=16384")
